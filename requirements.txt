# Kept deliberately small. Every package is one more thing that can fail to
# install, or be blocked, on a managed Windows machine, so the .docx reader, the
# .env reader and the workbook writer are written against the standard library
# and openpyxl rather than python-docx, python-dotenv and pandas.
# tests/test_dependencies.py fails if anything else creeps back in.

# --- Retrieval and generation ------------------------------------------------
openai==1.40.0                   # Azure OpenAI: embeddings AND chat

# openai 1.40.0 passes `proxies=` to httpx, which httpx 0.28 removed. Left
# unpinned, the Azure client raises TypeError at construction before any request
# is made. Lift this pin only together with an openai upgrade.
httpx>=0.27,<0.28

# openai declares `jiter<1`, so a fresh install takes the newest build. On a
# managed Windows machine jiter 0.17.0's native DLL is blocked by Application
# Control -- `import openai` then dies with "DLL load failed while importing
# jiter: An Application Control policy has blocked this file", which reads like
# a corrupt install rather than a policy decision, and takes every Azure call
# with it. 0.16.0 loads on the same machine. lxml 6 failed the same way before
# it left with python-docx. Verify with `python -c "from openai import AzureOpenAI"`
# after any bump.
jiter>=0.4.0,<0.17

# Held below 2: 1.26 is the build known to load where Application Control runs,
# and a numpy 2 wheel is new native code. Verify with `python -c "import numpy"`
# on that machine before moving the pin.
numpy>=1.26,<2                   # cosine ranking via matmul
pydantic==2.8.2                  # response schemas: intent scorer, Tier-2 Actor/Judge

# --- Excel I/O ---------------------------------------------------------------
openpyxl==3.1.5                  # reads .xlsx input; writes every workbook

# --- Dev ---------------------------------------------------------------------
pytest==8.3.2
pytest-asyncio==0.23.8
